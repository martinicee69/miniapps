    <script>
        // --- CONSTANTS ---
        const CONTRACT_ADDRESS = "0x93f31e00e6c8f91d46cfa3216e949259b079d959";
        // Using a mock address for the connected user's wallet
        const MOCK_WALLET_ADDRESS = "0xPhantomDApp1234567890ConnectedWallet"; 
        const MINT_PRICE = 0.05; // ETH per NFT
        
        // --- DOM Elements ---
        const connectButton = document.getElementById('connect-button');
        const addressDisplay = document.getElementById('address-display');
        const simulationWarning = document.getElementById('simulation-warning');
        const loadingSpinner = document.getElementById('loading-spinner');
        const mintButton = document.getElementById('mint-button');
        const mintAmountInput = document.getElementById('mint-amount');
        const mintStatus = document.getElementById('mint-status');
        const nftCountSpan = document.getElementById('nft-count');
        const gfsBalanceSpan = document.getElementById('gfs-balance');
        const copyNotification = document.getElementById('copy-notification');
        const walletAddressCode = document.getElementById('wallet-address-code');
        const contractAddressCode = document.getElementById('contract-address-code');

        // --- State ---
        let mockNftCount = 0;
        let mockGfsBalance = 500000;
        
        // Initialize contract address text content
        contractAddressCode.textContent = CONTRACT_ADDRESS;


        /**
         * Utility function to format numbers.
         * @param {number} num 
         * @returns {string}
         */
        const formatNumber = (num) => {
            return num.toLocaleString('en-US');
        }

        /**
         * Handles copying text to clipboard and showing a temporary notification.
         * @param {string} textToCopy 
         */
        function copyToClipboard(textToCopy) {
            // Use execCommand('copy') for better iframe compatibility
            const el = document.createElement('textarea');
            el.value = textToCopy;
            el.setAttribute('readonly', '');
            el.style.position = 'absolute';
            el.style.left = '-9999px';
            document.body.appendChild(el);
            el.select();
            try {
                document.execCommand('copy');
                copyNotification.classList.remove('hidden', 'opacity-0');
                copyNotification.classList.add('opacity-100');
                setTimeout(() => {
                    copyNotification.classList.remove('opacity-100');
                    copyNotification.classList.add('opacity-0');
                    setTimeout(() => copyNotification.classList.add('hidden'), 500);
                }, 1500);
            } catch (err) {
                console.error('Failed to copy text: ', err);
            }
            document.body.removeChild(el);
        }

        /**
         * Copies the mock wallet address to the clipboard.
         */
        function copyMyAddress() {
            copyToClipboard(MOCK_WALLET_ADDRESS);
            console.log("Mock Wallet Address Copied.");
        }

        /**
         * Copies the official contract address to the clipboard.
         */
        function copyContractAddress() {
            copyToClipboard(CONTRACT_ADDRESS);
            console.log("Contract Address Copied.");
        }


        /**
         * Simulates connecting to a Web3 wallet.
         */
        async function connectWallet() {
            // Disable button and show loading spinner
            connectButton.disabled = true;
            connectButton.innerHTML = `Connecting...`;
            loadingSpinner.style.display = 'block';

            // Simulate network latency (2 seconds delay)
            await new Promise(resolve => setTimeout(resolve, 2000));

            // Successful connection simulation
            try {
                // 1. Update the UI to show the connected state
                connectButton.classList.add('hidden');
                
                // Update address display with current mock data
                walletAddressCode.textContent = MOCK_WALLET_ADDRESS;
                gfsBalanceSpan.textContent = formatNumber(mockGfsBalance);
                nftCountSpan.textContent = mockNftCount;

                addressDisplay.classList.remove('hidden');
                simulationWarning.classList.remove('hidden');
                
                // Enable Minting button
                mintButton.disabled = false;
                updateMintTotal();

                // 2. Log success (for console review)
                console.log("Wallet connected successfully (Simulation)");

            } catch (error) {
                console.error("Wallet connection failed (Simulation):", error);
                connectButton.innerHTML = "Connection Failed. Try Again.";
                connectButton.classList.remove('hidden'); // Show button again
                connectButton.classList.add('bg-red-900', 'border-red-500');
                
            } finally {
                loadingSpinner.style.display = 'none';
                connectButton.disabled = false;
            }
        }
        
        /**
         * Updates the text on the mint button based on the selected amount.
         */
        function updateMintTotal() {
            const amount = parseInt(mintAmountInput.value, 10);
            if (isNaN(amount) || amount < 1) {
                mintAmountInput.value = 1;
                updateMintTotal();
                return;
            }
            // Max mint limit
            if (amount > 5) {
                mintAmountInput.value = 5;
                updateMintTotal();
                return;
            }
            
            const totalEth = (amount * MINT_PRICE).toFixed(2);
            mintButton.textContent = `Mint ${amount} NFT${amount > 1 ? 's' : ''} (Total: ${totalEth} ETH)`;
            mintStatus.textContent = ''; // Clear status on change
        }

        /**
         * Simulates the NFT minting transaction.
         */
        async function mintNFT() {
            if (mintButton.disabled) return;
            
            const amount = parseInt(mintAmountInput.value, 10);

            if (amount + mockNftCount > 5) {
                mintStatus.textContent = `Mint limit exceeded. You can only mint ${5 - mockNftCount} more NFTs.`;
                mintStatus.classList.remove('text-green-400', 'text-yellow-400');
                mintStatus.classList.add('text-red-500');
                return;
            }
            
            // Disable button and show status
            mintButton.disabled = true;
            mintStatus.textContent = 'Transaction initiating... (Simulating gas fee)';
            mintStatus.classList.remove('text-red-500');
            mintStatus.classList.add('text-yellow-400');

            // Simulate transaction delay
            await new Promise(resolve => setTimeout(resolve, 3000));
            
            // Successful mint simulation
            mockNftCount += amount;
            mockGfsBalance -= (amount * 1000); // Simulate token burn/fee
            
            // Update UI
            nftCountSpan.textContent = mockNftCount;
            gfsBalanceSpan.textContent = formatNumber(mockGfsBalance);
            
            mintStatus.textContent = `Success! Minted ${amount} Slayer NFT${amount > 1 ? 's' : ''}. Total held: ${mockNftCount}.`;
            mintStatus.classList.remove('text-yellow-400');
            mintStatus.classList.add('text-green-400');
            
            // Re-enable button
            mintButton.disabled = false;
            updateMintTotal();

            console.log(`Successfully minted ${amount} NFTs. New NFT count: ${mockNftCount}`);
        }
        
        // Expose functions globally
        window.connectWallet = connectWallet;
        window.mintNFT = mintNFT;
        window.updateMintTotal = updateMintTotal;
        window.copyMyAddress = copyMyAddress;
        window.copyContractAddress = copyContractAddress;

    </script>
